11def is_grayscale_frame(frame_bgr: np.ndarray) -> bool: 

12 """Heuristic: a 3-channel image that's actually grayscale (common for old photos saved 

13 as BGR/RGB with all channels equal, or scanned B&W) has near-zero difference between its 

14 B/G/R channels across the whole image. Downsamples first -- only the mean matters, and a 

15 small sample is far cheaper than scanning a full-resolution frame.""" 

16 small = cv2.resize(frame_bgr, (64, 64), interpolation=cv2.INTER_AREA).astype( 

17 np.float32 

18 ) 

19 b, g, r = small[..., 0], small[..., 1], small[..., 2] 

20 diff = (np.abs(b - g) + np.abs(g - r) + np.abs(b - r)) / 3.0 

21 return float(diff.mean()) < GRAYSCALE_CHANNEL_DIFF_THRESHOLD

24def crop_region(frame: np.ndarray, x1: int, y1: int, x2: int, y2: int) -> np.ndarray: 

25 """Crop an arbitrary rectangle, clamped to frame bounds.""" 

26 h, w = frame.shape[:2] 

27 x1, x2 = sorted((max(0, min(x1, w)), max(0, min(x2, w)))) 

28 y1, y2 = sorted((max(0, min(y1, h)), max(0, min(y2, h)))) 

29 return frame[y1:y2, x1:x2]

32def face_crop_bounds( 

33 box: tuple[int, int, int, int], 

34 frame_shape: tuple[int, int], 

35 padding_ratio: float = 0.1, 

36) -> tuple[int, int, int, int]: 

37 """Return a detector crop with scale-relative context, using exclusive bounds. 

38 

39 The Caffe age model is sensitive to how much surrounding context occupies its fixed 227x227 

40 input. A fixed pixel margin makes that context dominate small faces and disappear around 

41 large faces, so keep the framing proportionate to the detected face instead. 

42 """ 

43 x1, y1, x2, y2 = box 

44 frame_height, frame_width = frame_shape 

45 face_width, face_height = max(0, x2 - x1), max(0, y2 - y1) 

46 padding = max(1, round(max(face_width, face_height) * padding_ratio)) 

47 return ( 

48 max(0, x1 - padding), 

49 max(0, y1 - padding), 

50 min(frame_width, x2 + padding), 

51 min(frame_height, y2 + padding), 

52 )

65def _adjust_contrast(img: np.ndarray, amount: float) -> np.ndarray: 

66 """Apply contrast correction via the classic parametric formula.""" 

67 c = ( 

68 amount * 2.55 

69 ) # slider -100..100 -> classic contrast-correction-factor's -255..255 

70 factor = (259.0 * (c + 255.0)) / (255.0 * (259.0 - c)) 

71 return factor * (img - 128.0) + 128.0

74def _adjust_tone_region(img_bgr: np.ndarray, amount: float, region: str) -> np.ndarray: 

75 """Shift highlights or shadows via a luminance-weighted mask in HSV's V channel. 

76 Positive `amount` brightens highlights / lifts shadows (Lightroom convention).""" 

77 hsv = cv2.cvtColor( 

78 np.clip(img_bgr, 0, 255).astype(np.uint8), cv2.COLOR_BGR2HSV 

79 ).astype(np.float32) 

80 v = hsv[..., 2] 

81 if region == "highlights": 

82 mask = np.clip((v - 128.0) / 127.0, 0.0, 1.0) 

83 else: 

84 mask = np.clip((128.0 - v) / 128.0, 0.0, 1.0) 

85 hsv[..., 2] = np.clip(v + (amount / 100.0) * 50.0 * mask, 0, 255) 

86 return cv2.cvtColor(hsv.astype(np.uint8), cv2.COLOR_HSV2BGR).astype(np.float32)

89def _adjust_black_point(img: np.ndarray, amount: float) -> np.ndarray: 

90 """Shift and stretch the shadow point (rescales to compensate).""" 

91 bp = np.clip((amount / 100.0) * 60.0, -60.0, 250.0) 

92 return (img - bp) * (255.0 / max(255.0 - bp, 1.0))

95def _adjust_saturation( 

96 img_bgr: np.ndarray, amount: float, vibrance: bool = False 

97) -> np.ndarray: 

98 """Adjust saturation or skin-preserving vibrance in HSV color space.""" 

99 hsv = cv2.cvtColor( 

100 np.clip(img_bgr, 0, 255).astype(np.uint8), cv2.COLOR_BGR2HSV 

101 ).astype(np.float32) 

102 s = hsv[..., 1] 

103 if vibrance: 

104 # Boost low-saturation pixels more than already-saturated ones (protects skin tones). 

105 s = s + (amount / 100.0) * 60.0 * (1.0 - s / 255.0) 

106 else: 

107 s = s * (1.0 + amount / 100.0) 

108 hsv[..., 1] = np.clip(s, 0, 255) 

109 return cv2.cvtColor(hsv.astype(np.uint8), cv2.COLOR_HSV2BGR).astype(np.float32)

112def _adjust_sharpness(img_bgr: np.ndarray, amount: float) -> np.ndarray: 

113 """Classic unsharp mask -- small-radius blur subtracted back out to boost edge contrast.""" 

114 blurred = cv2.GaussianBlur(img_bgr, (0, 0), sigmaX=1.5) 

115 return img_bgr + (amount / 100.0) * 1.5 * (img_bgr - blurred)

118def _adjust_definition(img_bgr: np.ndarray, amount: float) -> np.ndarray: 

119 """'Clarity'-style local contrast: large-radius unsharp mask on the LAB lightness channel 

120 only, so it boosts midtone structure without shifting color.""" 

121 lab = cv2.cvtColor( 

122 np.clip(img_bgr, 0, 255).astype(np.uint8), cv2.COLOR_BGR2LAB 

123 ).astype(np.float32) 

124 L = lab[..., 0] 

125 blurred = cv2.GaussianBlur(L, (0, 0), sigmaX=12.0) 

126 lab[..., 0] = np.clip(L + (amount / 100.0) * 1.2 * (L - blurred), 0, 255) 

127 return cv2.cvtColor(lab.astype(np.uint8), cv2.COLOR_LAB2BGR).astype(np.float32)
