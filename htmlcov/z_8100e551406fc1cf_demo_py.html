42def _shared_analysis_slot() -> threading.BoundedSemaphore: 

43 """Return the process-wide analysis semaphore, creating it on first use.""" 

44 holder = sys.modules.setdefault( 

45 ANALYSIS_SLOT_HOLDER, types.ModuleType(ANALYSIS_SLOT_HOLDER) 

46 ) 

47 # dict.setdefault is atomic, so concurrent imports still agree on one semaphore. 

48 return vars(holder).setdefault("analysis_slot", threading.BoundedSemaphore(1))

98@contextmanager 

99def analysis_slot(on_wait: Callable[[], object]) -> Iterator[None]: 

100 """Hold the process-wide analysis slot, calling on_wait once if another analysis holds it.""" 

101 if not _ANALYSIS_SLOT.acquire(blocking=False): 

102 on_wait() 

103 if not _ANALYSIS_SLOT.acquire(timeout=DEMO_ANALYSIS_WAIT_SECONDS): 

104 raise DemoBusyError("The demo is busy. Try again in a minute.") 

105 try: 

106 yield 

107 finally: 

108 _ANALYSIS_SLOT.release()
